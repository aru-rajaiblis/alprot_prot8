SinglyLinkedListNode* removeDuplicates(SinglyLinkedListNode* llist) {
    if (llist == nullptr) {
        return llist;
    }
    
    // Disamakan tipenya dengan parameter fungsi
    SinglyLinkedListNode* kurakura = llist; 
    
    while (kurakura != nullptr && kurakura->next != nullptr) {    
        SinglyLinkedListNode* temp = kurakura->next;
        
        if (kurakura->data == temp->data) {
            kurakura->next = kurakura->next->next;
            delete temp;
        } else {
            kurakura = kurakura->next; 
        }
    } 
    
    return llist;
}
